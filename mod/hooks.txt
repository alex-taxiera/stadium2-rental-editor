# Hooks into the stock game. See tools/mod/modtool.py.
#
# kind  stock function  target in the mod fragment

# Fragment 14's setup, run once when the entry screen starts. Loading the mod here
# puts it in memory before any other hook can fire, and before the screen claims
# the rest of the main pool for itself, which is also why the mod loads what it
# needs of the game's art here.
boot    func_8130251C   Mod_EntrySetup

# Draws the ENTER / EXCHANGE menu beside the detail panel.
jump    func_8130DA04   Mod_DrawEnterMenu

# Works out the ENTER menu's box, for drawing it and for the screen region it covers.
jump    func_8130D1FC   Mod_EnterMenuRect

# Handles input while the ENTER menu is open.
jump    func_8130EC9C   Mod_EnterMenuInput

# Sets up the ITEM list from the mask of items on offer.
jump    func_81306B64   Mod_ItemListInit

# Asks the menu shown once a team is chosen (OK / OK to Register / ...) for its
# result and acts on it. Runs from the mod only so that its call to the menu can be
# pointed at Mod_PollTeamMenu (mod/asm_patches.txt); its switch leads back into the
# stock body, hence `detour`.
detour  func_81303648   orig_func_81303648

# Sets up the team a side is about to choose, with its name.
jump    func_81302EE4   Mod_PartyInit

# Draws one entry of a popup menu.
jump    func_813114B8   Mod_DrawMenuRow

# Draws a popup menu.
jump    func_81311650   Mod_DrawMenu
